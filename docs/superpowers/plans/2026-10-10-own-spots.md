**Goal:** Let the user add a spot anywhere in the US from a pin on a map or the device's location, name it, and save it on the device, with its NWS cell, time zone, nearest tide station and nearest gauge worked out and disclosed.

**Architecture:** A baked list of NOAA stations ships with the app and is loaded only with the add screen. Saved spots are whole `Spot` objects in one localStorage key, validated on read. The map is Leaflet on OpenStreetMap tiles in one component; the add screen is lazy-loaded.

**Tech Stack:** Vite, React 19, strict TypeScript, Vitest, plain CSS, Leaflet 1.9.4 (new), OpenStreetMap tiles.

```ts
// src/data/geo.ts
// Distance and direction over the Earth's surface, for saying where a
// station stands from a spot. No React, no fetch.

export interface LatLon {
  lat: number
  lon: number
}

const EARTH_RADIUS_MI = 3958.8
const toRadians = (degrees: number) => (degrees * Math.PI) / 180
const toDegrees = (radians: number) => (radians * 180) / Math.PI

/** Great-circle distance in miles, by the haversine formula. */
export function distanceMi(a: LatLon, b: LatLon): number {
  const dLat = toRadians(b.lat - a.lat)
  const dLon = toRadians(b.lon - a.lon)
  const h =
    Math.sin(dLat / 2) ** 2 +
    Math.cos(toRadians(a.lat)) *
      Math.cos(toRadians(b.lat)) *
      Math.sin(dLon / 2) ** 2
  return 2 * EARTH_RADIUS_MI * Math.asin(Math.sqrt(h))
}

/** The initial bearing from a to b in degrees, 0 north and 90 east. */
export function bearingDeg(a: LatLon, b: LatLon): number {
  const lat1 = toRadians(a.lat)
  const lat2 = toRadians(b.lat)
  const dLon = toRadians(b.lon - a.lon)
  const y = Math.sin(dLon) * Math.cos(lat2)
  const x =
    Math.cos(lat1) * Math.sin(lat2) -
    Math.sin(lat1) * Math.cos(lat2) * Math.cos(dLon)
  return (toDegrees(Math.atan2(y, x)) + 360) % 360
}

const WORDS = [
  'north',
  'northeast',
  'east',
  'southeast',
  'south',
  'southwest',
  'west',
  'northwest',
]

/** The 8-point compass word for a bearing in degrees. */
export function directionWord(degrees: number): string {
  const turn = ((degrees % 360) + 360) % 360
  return WORDS[Math.round(turn / 45) % 8]
}

/**
 * Miles as the captions say them: one decimal under 10 mi, whole miles
 * from 10, matching "8.6 mi", "0.2 mi" and "31 mi" on the hard-coded spots.
 */
export function roundMiles(mi: number): number {
  const tenths = Math.round(mi * 10) / 10
  return tenths < 10 ? tenths : Math.round(mi)
}
```

```js
// scripts/bake-stations.mjs
// Fetch NOAA's station lists and write the trimmed list the app ships.
// Run by hand: node scripts/bake-stations.mjs. No dependencies.
//
// One row per station: [id, name, lat, lon, kind], where kind is
//   R   harmonic prediction station (publishes a 6-minute curve)
//   S   subordinate prediction station (highs and lows only)
//   W   water-level station with no predictions (the Great Lakes, mostly)
//   RW  or SW: a prediction station that also has a gauge
// Coordinates are kept to four decimals, about 36 ft.

import { writeFile } from 'node:fs/promises'

const BASE = 'https://api.tidesandcurrents.noaa.gov/mdapi/prod/webapi'
const OUT = new URL('../src/data/stations.json', import.meta.url)

async function list(type) {
  const response = await fetch(`${BASE}/stations.json?type=${type}`)
  if (!response.ok) throw new Error(`NOAA: HTTP ${response.status} for ${type}`)
  const { stations } = await response.json()
  return stations
}

const round4 = (n) => Math.round(n * 1e4) / 1e4

const predictions = await list('tidepredictions')
const gauges = await list('waterlevels')
const gaugeIds = new Set(gauges.map((s) => s.id))
const predictionIds = new Set(predictions.map((s) => s.id))

const rows = []
for (const s of predictions) {
  if (s.type !== 'R' && s.type !== 'S') {
    throw new Error(`NOAA: station ${s.id} has type ${s.type}`)
  }
  const kind = gaugeIds.has(s.id) ? `${s.type}W` : s.type
  rows.push([s.id, s.name.trim(), round4(s.lat), round4(s.lng), kind])
}
for (const s of gauges) {
  if (predictionIds.has(s.id)) continue
  rows.push([s.id, s.name.trim(), round4(s.lat), round4(s.lng), 'W'])
}
rows.sort((a, b) => (a[0] < b[0] ? -1 : a[0] > b[0] ? 1 : 0))

// One row per line, so a re-bake diffs by station.
const text = `[\n${rows.map((row) => JSON.stringify(row)).join(',\n')}\n]\n`
await writeFile(OUT, text)
console.log(`${rows.length} stations written to ${OUT.pathname}`)
```

```ts
// src/data/stations.ts
// NOAA's stations, baked into src/data/stations.json by
// scripts/bake-stations.mjs, and which of them a spot uses.

import type { Station } from '../spot.ts'
import { bearingDeg, directionWord, distanceMi, roundMiles } from './geo.ts'
import type { LatLon } from './geo.ts'

/**
 * R: harmonic prediction station, with a 6-minute curve. S: subordinate,
 * highs and lows only. W: a gauge with no predictions. RW or SW: a
 * prediction station with a gauge.
 */
export type StationKind = 'R' | 'S' | 'W' | 'RW' | 'SW'

export type StationRow = [
  id: string,
  name: string,
  lat: number,
  lon: number,
  kind: StationKind,
]

const KINDS = new Set<string>(['R', 'S', 'W', 'RW', 'SW'])

export function isStationRows(data: unknown): data is StationRow[] {
  return (
    Array.isArray(data) &&
    data.every(
      (row: unknown) =>
        Array.isArray(row) &&
        row.length === 5 &&
        typeof row[0] === 'string' &&
        typeof row[1] === 'string' &&
        typeof row[2] === 'number' &&
        typeof row[3] === 'number' &&
        KINDS.has(row[4] as string),
    )
  )
}

/** The baked list, fetched with the add screen rather than every open. */
export async function loadStations(): Promise<StationRow[]> {
  const module = await import('./stations.json')
  const rows: unknown = module.default
  if (!isStationRows(rows)) throw new Error('stations.json is not a list')
  return rows
}

const predicts = (row: StationRow) => row[4] !== 'W'
const gaugeWithCurve = (row: StationRow) => row[4] === 'RW'

function nearest(
  point: LatLon,
  rows: StationRow[],
  fits: (row: StationRow) => boolean,
): StationRow {
  let best: StationRow | null = null
  let bestMi = Infinity
  for (const row of rows) {
    if (!fits(row)) continue
    const mi = distanceMi(point, { lat: row[2], lon: row[3] })
    if (mi < bestMi) {
      best = row
      bestMi = mi
    }
  }
  if (!best) throw new Error('no station of that kind')
  return best
}

function asStation(point: LatLon, row: StationRow): Station {
  const at = { lat: row[2], lon: row[3] }
  return {
    id: row[0],
    name: row[1],
    distanceMi: roundMiles(distanceMi(point, at)),
    direction: directionWord(bearingDeg(point, at)),
    type: row[4].startsWith('R') ? 'harmonic' : 'subordinate',
  }
}

/**
 * The nearest prediction station, and the nearest station that is both
 * harmonic and a gauge. The gauge's readings are set against its own
 * predicted curve, so a gauge with no curve, as on the Great Lakes, would
 * have nothing to deviate from and is never chosen. Where the tide station
 * is itself such a gauge, it is the gauge.
 */
export function nearestStations(
  point: LatLon,
  rows: StationRow[],
): { tideStation: Station; gauge: Station } {
  const tide = nearest(point, rows, predicts)
  const gauge = gaugeWithCurve(tide) ? tide : nearest(point, rows, gaugeWithCurve)
  return { tideStation: asStation(point, tide), gauge: asStation(point, gauge) }
}
```

```ts
/** The point is outside every NWS grid: no forecast there, ever. */
export class NwsOutsideError extends Error {}

export interface NwsCell {
  office: string
  gridX: number
  gridY: number
  /** IANA zone, the one every time at the spot is shown in. */
  timeZone: string
}

const round4 = (n: number) => Math.round(n * 1e4) / 1e4

/** NWS takes four decimals at most and redirects anything longer. */
export function pointsUrl(lat: number, lon: number): string {
  return `https://api.weather.gov/points/${round4(lat)},${round4(lon)}`
}

export function parsePoints(json: unknown): NwsCell {
  const properties = (json as { properties?: Record<string, unknown> } | null)
    ?.properties
  if (!properties) throw new Error('NWS: response has no properties')
  const { gridId, gridX, gridY, timeZone } = properties
  if (
    typeof gridId !== 'string' ||
    typeof gridX !== 'number' ||
    typeof gridY !== 'number' ||
    typeof timeZone !== 'string'
  ) {
    throw new Error('NWS: response has no grid cell')
  }
  return { office: gridId, gridX, gridY, timeZone }
}

/** The forecast cell containing a point. Throws NwsOutsideError on a 404. */
export async function fetchPoints(lat: number, lon: number): Promise<NwsCell> {
  const response = await fetch(pointsUrl(lat, lon), {
    headers: { Accept: 'application/geo+json' },
  })
  if (response.status === 404) {
    throw new NwsOutsideError('NWS: no forecast for this point')
  }
  if (!response.ok) throw new Error(`NWS: HTTP ${response.status}`)
  return parsePoints(await response.json())
}

const FT_PER_M = 3.28084

/** A cell's average height in whole feet, from the gridpoint response. */
export function parseElevationFt(json: unknown): number {
  const properties = (json as { properties?: Record<string, unknown> } | null)
    ?.properties
  const elevation = properties?.elevation as
    | { unitCode?: unknown; value?: unknown }
    | undefined
  if (!elevation || typeof elevation.value !== 'number') {
    throw new Error('NWS: response has no elevation')
  }
  if (elevation.unitCode !== 'wmoUnit:m') {
    throw new Error(`NWS: elevation is in ${String(elevation.unitCode)}`)
  }
  return Math.round(elevation.value * FT_PER_M)
}

export async function fetchElevationFt(
  office: string,
  gridX: number,
  gridY: number,
): Promise<number> {
  const response = await fetch(nwsUrl(office, gridX, gridY), {
    headers: { Accept: 'application/geo+json' },
  })
  if (!response.ok) throw new Error(`NWS: HTTP ${response.status}`)
  return parseElevationFt(await response.json())
}
```

```ts
// src/data/resolve.ts
// What a pin on the map becomes: the NWS cell and zone, the nearest
// stations, and whether the tide is shown to start with. The requests are
// handed in, so this can be tested without a network.

import { NwsOutsideError } from './nws.ts'
import type { NwsCell } from './nws.ts'
import { nearestStations } from './stations.ts'
import type { StationRow } from './stations.ts'
import type { Spot } from '../spot.ts'

/** Tides are shown to start with when the station is this near... */
export const TIDES_WITHIN_MI = 25
/** ...and the cell's average height is under this. */
export const TIDES_BELOW_FT = 300

/**
 * Distance alone is not enough: a 4,000 ft peak can be 6 mi from a
 * station. Elevation alone is not enough either: a lake shore is low and
 * hundreds of miles from any tide.
 */
export function defaultTidesShown(
  tideDistanceMi: number,
  cellElevationFt: number,
): boolean {
  return tideDistanceMi <= TIDES_WITHIN_MI && cellElevationFt < TIDES_BELOW_FT
}

/** Everything a spot has apart from what the user gives it. */
export type SpotFacts = Omit<Spot, 'id' | 'name'>

export interface ResolveIo {
  fetchPoints: (lat: number, lon: number) => Promise<NwsCell>
  fetchElevationFt: (
    office: string,
    gridX: number,
    gridY: number,
  ) => Promise<number>
  stations: StationRow[]
}

/** outside: NWS has no forecast there. unreachable: it could not be asked. */
export type ResolveFailure = 'outside' | 'unreachable'

export type ResolveResult =
  | { ok: true; facts: SpotFacts }
  | { ok: false; reason: ResolveFailure }

const round4 = (n: number) => Math.round(n * 1e4) / 1e4

export async function resolveSpot(
  lat: number,
  lon: number,
  io: ResolveIo,
): Promise<ResolveResult> {
  const at = { lat: round4(lat), lon: round4(lon) }
  try {
    const cell = await io.fetchPoints(at.lat, at.lon)
    const elevationFt = await io.fetchElevationFt(
      cell.office,
      cell.gridX,
      cell.gridY,
    )
    const { tideStation, gauge } = nearestStations(at, io.stations)
    return {
      ok: true,
      facts: {
        ...at,
        timeZone: cell.timeZone,
        // No source gives the point's own height; issue #9 is where that goes.
        elevationFt: null,
        nws: {
          office: cell.office,
          gridX: cell.gridX,
          gridY: cell.gridY,
          elevationFt,
        },
        tideStation,
        gauge,
        tidesShown: defaultTidesShown(tideStation.distanceMi, elevationFt),
      },
    }
  } catch (error) {
    return {
      ok: false,
      reason: error instanceof NwsOutsideError ? 'outside' : 'unreachable',
    }
  }
}
```

```ts
// The spots the user added, saved on the device as whole Spot objects under
// one key. Every access is wrapped, as the cache's are, so a browser with
// storage disabled still works. Each entry is checked on read: a damaged
// one, or one from another build, is dropped rather than break every open.

import type { Spot, Station } from '../spot.ts'

export const SAVED_KEY = 'tideline:spots:v1'

function isStation(value: unknown): value is Station {
  if (typeof value !== 'object' || value === null) return false
  const s = value as Record<string, unknown>
  return (
    typeof s.id === 'string' &&
    typeof s.name === 'string' &&
    typeof s.distanceMi === 'number' &&
    typeof s.direction === 'string' &&
    (s.type === 'harmonic' || s.type === 'subordinate')
  )
}

export function isSpot(value: unknown): value is Spot {
  if (typeof value !== 'object' || value === null) return false
  const s = value as Record<string, unknown>
  const nws = s.nws as Record<string, unknown> | null | undefined
  return (
    typeof s.id === 'string' &&
    typeof s.name === 'string' &&
    typeof s.lat === 'number' &&
    typeof s.lon === 'number' &&
    typeof s.timeZone === 'string' &&
    (s.elevationFt === null || typeof s.elevationFt === 'number') &&
    typeof nws === 'object' &&
    nws !== null &&
    typeof nws.office === 'string' &&
    typeof nws.gridX === 'number' &&
    typeof nws.gridY === 'number' &&
    typeof nws.elevationFt === 'number' &&
    isStation(s.tideStation) &&
    isStation(s.gauge) &&
    typeof s.tidesShown === 'boolean'
  )
}

export function readSavedSpots(): Spot[] {
  try {
    const text = localStorage.getItem(SAVED_KEY)
    if (text === null) return []
    const list: unknown = JSON.parse(text)
    if (!Array.isArray(list)) return []
    return list.filter(isSpot)
  } catch {

}

/** An id no hard-coded spot has, from the time it was made. */
export function newSpotId(now: number = Date.now()): string {
  return `spot-${now.toString(36)}`
```

```ts
  heading: 'New spot',

  remove: 'Remove this spot',
  removeYes: 'Yes, remove',
  removeNo: 'Keep',

function stationWords(station: Station): string {
  return `NOAA ${station.id} ${station.name}, ${station.distanceMi} mi ${station.direction}`
}

/** The three lines under "What this spot gets". */
export function factsLines(facts: SpotFacts): {
  weather: string
  tide: string
  gauge: string
} {
  const { nws, tideStation, gauge } = facts
  const curve =
    tideStation.type === 'harmonic'
      ? '6-minute curve.'
      : 'Highs and lows only; the curve will be interpolated.'
  return {
    weather: `Weather: NWS cell ${nws.office} ${nws.gridX},${nws.gridY}, ${feet(nws.elevationFt)}, ${facts.timeZone}.`,
    tide: `Tide: ${stationWords(tideStation)}. ${curve}`,
    gauge:
      gauge.id === tideStation.id
        ? 'Gauge: its own.'
        : `Gauge: ${stationWords(gauge)}.`,
  }
}

/**
 * Why the tide is off to start with, or null where it is on. Where both
 * reasons hold, the distance: a far station is the plainer fact.
 */
export function tidesOffReason(facts: SpotFacts): string | null {
  if (facts.tidesShown) return null
  if (facts.tideStation.distanceMi > TIDES_WITHIN_MI) {
    return `Off to start: the nearest station is ${facts.tideStation.distanceMi} mi away.`
  }
  return `Off to start: the cell is ${feet(facts.nws.elevationFt)} up.`
}

export function failureWords(reason: ResolveFailure): string {
  switch (reason) {
    case 'outside':
      return 'NWS has no forecast here. Tideline covers the US.'
    case 'unreachable':
      return "Couldn't reach the National Weather Service. Try again in a minute."
  }
}

export function cleanName(raw: string): string {
  return raw.trim().slice(0, NAME_MAX)
}

export function canSave(
  pin: LatLon | null,
  name: string,
  result: ResolveResult | null,
): boolean {
  return pin !== null && cleanName(name) !== '' && result?.ok === true
}

export function removeQuestion(name: string): string {
  return `Remove ${name}?`
}
```

```tsx
// src/ui/SpotMap.tsx
import L from 'leaflet'
import 'leaflet/dist/leaflet.css'
import { useEffect, useRef } from 'react'
import type { LatLon } from '../data/geo.ts'
import { WORDS } from './addSpotView.ts'

const TILES = 'https://tile.openstreetmap.org/{z}/{x}/{y}.png'
const ATTRIBUTION =
  '&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors'
/** About 30 miles across a phone's width at mid latitudes. */
const ZOOM = 10
const PIN = { radius: 9, color: '#fff', weight: 2, fillColor: '#c8401f', fillOpacity: 1 }

interface SpotMapProps {
  /** Where the map opens. Read once. */
  center: LatLon
  pin: LatLon | null
  /** Somewhere to pan to when it changes, such as the device's position. */
  look: LatLon | null
  onPick: (at: LatLon) => void
}

/**
 * Leaflet on OpenStreetMap tiles, in a div it owns. A tap reports a point;
 * the pin is drawn where the parent says it is. A circle marker, not
 * Leaflet's image marker, so there is no icon file for the bundler to lose.
 */
export function SpotMap({ center, pin, look, onPick }: SpotMapProps) {
  const host = useRef<HTMLDivElement>(null)
  const map = useRef<L.Map | null>(null)
  const marker = useRef<L.CircleMarker | null>(null)
  // The latest handler, so the map's one listener never goes stale.
  const pick = useRef(onPick)
  pick.current = onPick

  useEffect(() => {
    if (!host.current) return
    const made = L.map(host.current).setView([center.lat, center.lon], ZOOM)
    L.tileLayer(TILES, { attribution: ATTRIBUTION, maxZoom: 19 }).addTo(made)
    made.on('click', (event: L.LeafletMouseEvent) => {
      pick.current({ lat: event.latlng.lat, lon: event.latlng.lng })
    })
    map.current = made
    return () => {
      made.remove()
      map.current = null
      marker.current = null
    }
    // The centre is where the map opens, not something it follows.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [])

  useEffect(() => {
    const made = map.current
    if (!made) return
    if (!pin) {
      marker.current?.remove()
      marker.current = null
      return
    }
    if (marker.current) {
      marker.current.setLatLng([pin.lat, pin.lon])
    } else {
      marker.current = L.circleMarker([pin.lat, pin.lon], PIN).addTo(made)
    }
  }, [pin])

  useEffect(() => {
    if (look && map.current) map.current.setView([look.lat, look.lon], ZOOM)
  }, [look])

  return (
    <div
      ref={host}
      className="map"
      role="application"
      aria-label={WORDS.mapLabel}
    />
  )
}
```

```tsx
  const [look, setLook] = useState<LatLon | null>(null)
  const [name, setName] = useState('')
  const [stations, setStations] = useState<StationRow[] | null>(null)
  const [result, setResult] = useState<ResolveResult | null>(null)
  const [attempt, setAttempt] = useState(0)
  const [tidesShown, setTidesShown] = useState<boolean | null>(null)
  const [locating, setLocating] = useState<'idle' | 'finding' | 'failed'>(
    'idle',
  )

  const tidesId = useId()

  useEffect(() => {
    let gone = false
    loadStations().then(
      (rows) => {
        if (!gone) setStations(rows)
      },
      () => {
        // The list did not load: the add screen cannot resolve anything.
        if (!gone) setStations([])
      },
    )
    return () => {
      gone = true
    }
  }, [])

  // Each pin, and each try-again, looks the pin up afresh.
  useEffect(() => {
    if (!pin || !stations) return
    let gone = false
    setResult(null)
    setTidesShown(null)
    resolveSpot(pin.lat, pin.lon, {
      fetchPoints,
      fetchElevationFt,
      stations,

      if (gone) return
      setResult(answer)
      if (answer.ok) setTidesShown(answer.facts.tidesShown)

    return () => {
      gone = true

      return

          lat: position.coords.latitude,
          lon: position.coords.longitude,

        setLocating('idle')
        setPin(at)
        setLook(at)
      },
      () => setLocating('failed'),
      { timeout: LOCATE_TIMEOUT, maximumAge: 60_000 },
    )
  }

    if (!pin || !result?.ok) return
    onSave({
      ...result.facts,
      id: newSpotId(),
      name: cleanName(name),

        />

                {WORDS.tryAgain}
              </button>

          className="spot-choice"

          {WORDS.save}

          {WORDS.cancel}
        </button>

    </section>
```

Replace `src/App.tsx` with:

```tsx
import { Suspense, lazy, useState } from 'react'
import { readSavedSpots, addSpot } from './data/savedSpots.ts'
import { SPOTS, allSpots, isSaved } from './spot.ts'
import type { Spot } from './spot.ts'
import { WORDS } from './ui/addSpotView.ts'
import { forgetSavedSpot, readSpot, writeSpot } from './ui/choices.ts'
import { InstallHint } from './ui/InstallHint.tsx'
import { SpotScreen } from './ui/SpotScreen.tsx'
import { SpotSwitcher } from './ui/SpotSwitcher.tsx'

// Leaflet and the station list come with it, the first time it is opened.
const AddSpot = lazy(() => import('./ui/AddSpot.tsx'))

export default function App() {
  const [saved, setSaved] = useState<Spot[]>(readSavedSpots)
  const [spot, setSpot] = useState<Spot>(() => readSpot(allSpots(saved)))
  const [adding, setAdding] = useState(false)
  const spots = allSpots(saved)

  const selectSpot = (next: Spot) => {
    writeSpot(next)
    setSpot(next)
  }
  const saveSpot = (next: Spot) => {
    addSpot(next)
    setSaved([...saved, next])
    selectSpot(next)
    setAdding(false)
    window.scrollTo(0, 0)
  }
  const removeSpot = () => {
    forgetSavedSpot(spot)
    setSaved(saved.filter((each) => each.id !== spot.id))
    selectSpot(SPOTS[0])
    window.scrollTo(0, 0)
  }

  if (adding) {
    return (
      <main>
        <h1>{WORDS.heading}</h1>
        <Suspense fallback={<p className="notice">{WORDS.loadingMap}</p>}>
          <AddSpot
            from={spot}
            onSave={saveSpot}
            onCancel={() => setAdding(false)}
          />
        </Suspense>
      </main>
    )
  }

  return (
    <main>
      <h1>{spot.name}</h1>
      <SpotSwitcher
        spots={spots}
        current={spot}
        onSelect={selectSpot}
        onAdd={() => setAdding(true)}
      />
      <InstallHint />
      {/* Keyed, so a switch mounts the screen afresh, as a first open does. */}
      <SpotScreen
        key={spot.id}
        spot={spot}
        onRemove={isSaved(spot) ? removeSpot : undefined}
      />
    </main>
  )
}
```
